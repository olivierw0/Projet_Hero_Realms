        +~Carte()

        +getNom() string

        +getFaction() FactionCarte
        +getDescription() string
        +getEffets() vector~Effet~
        +jouerCarte(joueurActif, adversaire) void

    %% Relations d'héritage
    Champion --|> Carte : hérite

    %% Relations de composition/agrégation
    Carte *-- TypeCarte : utilise
    Carte *-- FactionCarte : utilise
    Carte o-- Effet : contient 0..*
    
    Champion *-- TypeCarte : CHAMPION
    Champion *-- FactionCarte : utilise

    Effet *-- TypeEffet : utilise
    Effet *-- ConditionEffet : utilise

    ZoneDeCarte *-- TypeZone : utilise
    ZoneDeCarte o-- Carte : contient 0..*
    ZoneDeCarte --> Joueur : propriétaire 0..1
    ZoneDeCarte --> TablePartie : table 0..1

    Joueur o-- ZoneDeCarte : possède 1..*
    Joueur --> Carte : joue

    TablePartie o-- Joueur : contient 2..*
    TablePartie o-- ZoneDeCarte : zones_communes 0..*

    %% Relations d'utilisation
    Carte ..> Joueur : utilise dans jouerCarte()
    Effet ..> Joueur : utilise dans activerEffet()
    Champion ..> Joueur : utilise dans expend()
